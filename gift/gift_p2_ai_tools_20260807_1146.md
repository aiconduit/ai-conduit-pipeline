## 5️⃣ 無料API切り替え用コードスニペット

### 🔄 マルチプロバイダー対応テンプレート

```python
import os
import requests
from typing import Dict, Any

class FreeAIProvider:
    """無料AI API 統合クラス"""
    
    def __init__(self):
        self.providers = {
            "openrouter": {
                "url": "https://openrouter.ai/api/v1/chat/completions",
                "model": "meta-llama/llama-3.1-8b-instruct:free"
            },
            "groq": {
                "url": "https://api.groq.com/openai/v1/chat/completions",
                "model": "llama-3.1-8b-instant"
            },
            "cloudflare": {
                "url": f"https://api.cloudflare.com/client/v4/accounts/{os.getenv('CF_ACCOUNT_ID')}/ai/run/@cf/meta/llama-3.1-8b-instruct",
                "model": None
            }
        }
    
    def generate(self, prompt: str, provider: str = "groq") -> Dict[str, Any]:
        """テキスト生成を実行"""
        config = self.providers[provider]
        headers = {
            "Authorization": f"Bearer {os.getenv(provider.upper() + '_API_KEY')}",
            "Content-Type": "application/json"
        }
        
        payload = {
            "model": config["model"],
            "messages": [{"role": "user", "content": prompt}],
            "temperature": 0.7
        }
        
        response = requests.post(
            config["url"],
            headers=headers,
            json=payload
        )
        
        return response.json()
    
    def fallback_chain(self, prompt: str) -> str:
        """プロバイダー自動フォールバック"""
        for provider in ["groq", "openrouter", "cloudflare"]:
            try:
                result = self.generate(prompt, provider)
                return result["choices"][0]["message"]["content"]
            except Exception as e:
                print(f"{provider}失敗: {e}")
                continue
        return "全てのプロバイダーが失敗しました"

# 使用例
ai = FreeAIProvider()
result = ai.fallback_chain("AIの未来について教えて")
print(result)
```
